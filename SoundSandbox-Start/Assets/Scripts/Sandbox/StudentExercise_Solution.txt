void OnTriggerEnter(Collider other) { if (other.CompareTag(playerTag)) playerInside = true; }
void OnTriggerExit(Collider other)  { if (other.CompareTag(playerTag)) playerInside = false; }
void Update()
{
    float rate = playerInside ? risePerSecond : -fallPerSecond;
    adaptive.AddIntensity(rate * Time.deltaTime);
}
